ACTOR StrengthRune : RuneGiver 5100
{
	Game Doom
	Game Heretic
	Game Hexen
	SpawnID 180
	+COUNTITEM
	+NOGRAVITY
	+INVENTORY.AUTOACTIVATE
	Inventory.Icon STRRA0
	Inventory.MaxAmount 0
	Rune.Type DoubleDamage
	Inventory.PickupMessage "$PICKUP_STRENGTH"
	Inventory.PickupAnnouncerEntry "strength"
	States
	{
	Spawn:
		STRR ABCD 6
		Loop
	}
}

const int RUNE_PICKUP = 13239;
const int RUNE_STRENGTH = 0;
const int RUNE_RAGE = 1;
const int RUNE_DRAIN = 2;
const int RUNE_SPREAD = 3;
const int RUNE_RESISTANCE = 4;
const int RUNE_REGENERATION = 5;
const int RUNE_PROSPERITY = 6;
const int RUNE_REFLECTION = 7;
const int RUNE_HIGHJUMP = 8;
const int RUNE_HASTE = 9;

actor StrengthRuneGiver
{
	Activation THINGSPEC_ThingActs | THINGSPEC_ThingTargets
	+USESPECIAL
	States
	{
		Spawn:
			TNT1 A 0 NoDelay A_SetSpecial(84, RUNE_PICKUP, RUNE_STRENGTH)
			goto SpawnLoop
		SpawnLoop:
			STRR ABCD 6
			Loop
		Despawn:
			TNT1 A 1
			stop
	}
}
