actor SH_FireImp : SH_HereticImp
{
	DropItem "HealthBonus", 48
	States
	{
		Spawn:
			IMP2 ABCB 10 A_Look
			Loop
		See:
			IMP2 AABBCCBB 3 A_Chase
			Loop
		Melee:
			IMP2 DE 4 A_FaceTarget
			IMP2 F 6 A_CustomMeleeAttack(random[ImpMeAttack](5,12), "himp/attack", "himp/attack")
			Goto See
		Missile:
			IMP2 A 0 A_Jump(75, "Throw")
			IMP2 A 8 A_FaceTarget
			IMP2 B 6 A_SkullAttack(12)
			IMP2 CBAB 6
			Goto Missile+3
		Throw:
			IMP2 DE 4 A_FaceTarget
			IMP2 F 6 A_CustomComboAttack("SH_FireImpBall", 32, random[ImpMsAttack2](5,12), "himp/leaderattack")
			Goto See
		Pain:
			IMP2 G 3
			IMP2 G 3 A_Pain
			Goto See
		Death:
			IMP2 S 0 A_ChangeFlag("FLOATBOB", 0)
			IMP2 G 4 A_ImpDeath
			IMP2 H 5
			Wait
		XDeath:
			IMP2 S 0 A_ChangeFlag("FLOATBOB", 0)
			IMP2 S 5 A_ImpXDeath1
			IMP2 TU 5
			IMP2 V 5 A_Gravity
			IMP2 W 5 
			Wait
		Crash:
			IMP2 I 0 A_SpawnItemEx("FireImpChunk1", 0, 0, 0, frandom(-4.0, 4.0), frandom(-4.0,4.0), 9, 0, SXF_CLIENTSIDE, 0)
			IMP2 I 0 A_SpawnItemEx("FireImpChunk2", 0, 0, 0, frandom(-4.0, 4.0), frandom(-4.0,4.0), 9, 0, SXF_CLIENTSIDE, 0)
			IMP2 I 0 A_JumpIf(special == 666, "XCrash")
			IMP2 I 7
			IMP2 J 7 A_ScreamAndUnblock
			IMP2 K 7
			IMP2 L 10500
			goto Fade
		Fade:
			"####" "#" 1 A_FadeOut(0.05)
			loop
		XCrash:
			IMP2 X 7
			IMP2 Y 7 A_NoBlocking
			IMP2 Z 10500
			goto Fade
	}
}
