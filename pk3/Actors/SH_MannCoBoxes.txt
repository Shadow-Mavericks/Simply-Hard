actor Pink
{
	Radius 10
	Height 48
	+DROPOFF
	+CANPASS
	+THRUACTORS
	+LOOKALLAROUND
	var int user_bombsthrown;
	var int user_spamcount;
	States
	{
		Spawn:
			PINK C 0 NoDelay A_PlaySound("Pink/Mew", CHAN_BODY)
			PINK C 2 A_ChangeVelocity(0.0, 0.0, 10.0, CVF_REPLACE)
			PINK C 0 A_ChangeFlag("CORPSE", 1)
			PINK DEF 2
			goto JumpLoop
		JumpLoop:
			PINK G 2
			loop
		Crash:
			PINK FED 1
			PINK CB 2
			PINK A 12
			goto Laugh
		Laugh:
			PINK Y 0 A_PlaySound("Pink/LaughLong", CHAN_BODY)
			PINK YZYZYZYZ 4
			goto Throwing
		Throwing:
			PINK A 2 A_LookEx(LOF_NOJUMP | LOF_NOSOUNDCHECK, 0, 1024, 1024, 360.0)
			PINK BCDEFGHIJKLMN 2
			PINK O 0 A_PlaySound("Pink/Throw", CHAN_BODY)
			PINK O 2 A_PlaySound("Pink/BombThrow", CHAN_AUTO)
			PINK P 2 A_CustomMissile("PinkBomb", 32, 0, frandom(-45.0, 45.0), CMF_TRACKOWNER | CMF_OFFSETPITCH, frandom(4.0, 32.0))
			PINK P 0 A_SetUserVar("user_bombsthrown", user_bombsthrown+1)
			PINK Q 2
			PINK CBA 2
			PINK A 0 A_JumpIf(user_bombsthrown >= 100, "FinalBomb")
			PINK A 0 A_Jump(32, "SpamThrow")
			loop
			
		SpamThrow:
			PINK A 2 A_LookEx(LOF_NOJUMP | LOF_NOSOUNDCHECK, 0, 1024, 1024, 360.0)
			PINK BCDEFGHIJKLMN 2
			PINK O 0 A_PlaySound("Pink/Throw", CHAN_BODY)
			PINK O 2 A_PlaySound("Pink/BombThrow", CHAN_AUTO)
			PINK P 2 A_CustomMissile("PinkBomb", 32, 0, frandom(-45.0, 45.0), CMF_TRACKOWNER | CMF_OFFSETPITCH, frandom(4.0, 32.0))
			PINK P 0 A_SetUserVar("user_bombsthrown", user_bombsthrown+1)
			PINK Q 2 A_SetUserVar("user_spamcount", random(4, 10))
			goto SpamLoop
		
		SpamLoop:
			PINK D 0 A_LookEx(LOF_NOJUMP | LOF_NOSOUNDCHECK, 0, 1024, 1024, 360.0)
			PINK GHIJKLMN 1
			PINK O 0 A_PlaySound("Pink/Throw", CHAN_BODY)
			PINK O 1 A_PlaySound("Pink/BombThrow", CHAN_AUTO)
			PINK P 1 A_CustomMissile("PinkBomb", 32, 0, frandom(-45.0, 45.0), CMF_TRACKOWNER | CMF_OFFSETPITCH, frandom(4.0, 32.0))
			PINK P 0 A_SetUserVar("user_bombsthrown", user_bombsthrown+1)
			PINK P 0 A_SetUserVar("user_spamcount", user_spamcount-1)
			PINK Q 1
			PINK Q 0 A_JumpIf(user_spamcount <= 0, "EndThrow")
			loop
			
		EndThrow:
			PINK Q 8
			PINK CBA 2
			PINK A 0 A_JumpIf(user_bombsthrown >= 100, "FinalBomb")
			goto Laugh
			
		FinalBomb:
			PINK A 35
			PINK B 2
			PINK RSTU 2
			PINK U 40 A_PlaySound("Pink/LaughShort", CHAN_AUTO)
			PINK U 0 A_PlaySound("Pink/BadExplosion", CHAN_AUTO)
			PINK U 0 A_CustomMissile("PinkExplosionSpawner", 50, 0, 0, CMF_TRACKOWNER | CMF_ABSOLUTEANGLE | CMF_ABSOLUTEPITCH, 0)
			PINK U 0 A_CustomMissile("PinkExplosionSpawner", 50, 0, 90, CMF_TRACKOWNER | CMF_ABSOLUTEANGLE | CMF_ABSOLUTEPITCH, 0)
			PINK U 0 A_CustomMissile("PinkExplosionSpawner", 50, 0, 180, CMF_TRACKOWNER | CMF_ABSOLUTEANGLE | CMF_ABSOLUTEPITCH, 0)
			PINK U 0 A_CustomMissile("PinkExplosionSpawner", 50, 0, 270, CMF_TRACKOWNER | CMF_ABSOLUTEANGLE | CMF_ABSOLUTEPITCH, 0)
			PINK VW 1
			PINK X 175
			PINK X 0 A_ChangeFlag("NOGRAVITY", true)
			PINK X 0 A_ChangeVelocity(0.0, 0.0, 4.0, CVF_REPLACE)
			goto Fade
		Fade:
			PINK X 1 A_SetScale(scalex-0.1, scaley)
			PINK X 0 A_JumpIf(scalex <= 0.0, "Disappear")
			loop
		Disappear:
			TNT1 A 1
			stop
	}
}
