// Very basic spring. There's gotta be a better way to do it in Zandronum.

const float py = 1.4142135623730950488016887242097;

Actor SpringBase : SwitchableDecoration {
	-NOINTERACTION
	Height 24
	Radius 24
	Mass 0x7FFFFFFF
	States {
		Spawn:
			SPRI A 0 NoDelay ACS_NamedExecuteAlways("SpringScript") // Spring Script
			Goto Idle
		Idle:
			SPRI A 1 A_LookEx(LOF_NOSOUNDCHECK|LOF_DONTCHASEGOAL|LOF_NOJUMP, 0, Radius*py, 0, 360, 0) // Position check is done via async script
			Loop
		Active:
			SPRI A 1 A_ActiveSound
			SPRI BCDEFEEEDDDCCCBBBAAA 1 A_LookEx(LOF_NOSOUNDCHECK|LOF_DONTCHASEGOAL|LOF_NOJUMP, 0, Radius*py, 0, 360, 0) // Always be looking, just in case of other players
			Goto Idle
	}
}
