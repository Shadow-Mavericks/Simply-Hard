actor N_Cacodemon_Base : SH_BaseMonster
{
	Tag "Cacodemon"
	Species "Cacodemon"
	Health 400
	Radius 31
	Height 56
	Mass 400
	Speed 15
	PainChance 128
	PainChance "Mightyboot", 255
	BloodColor Blue
	+FLOAT
	+NOGRAVITY
	SeeSound "caco/sight"
	PainSound "caco/pain"
	DeathSound "caco/death"
	ActiveSound "caco/active"
	Obituary "$OB_CACO"
	HitObituary "$OB_CACOHIT"
	States
	{
		Spawn:
			HEAD A 0
			goto Spawn2
		Spawn2:
			HEAD A 10 A_Look
			Loop
		See:
			HEAD A 3 A_Chase
			Loop
		Missile:
			HEAD BC 4 A_FaceTarget
			HEAD D 4 Bright A_CustomMissile("CacodemonBall_Base", 24)
			HEAD BC 4 A_FaceTarget
			HEAD D 4 Bright A_CustomMissile("CacodemonBall_Base", 24, 0, random(-35, 35))
			HEAD BC 4 A_FaceTarget
			HEAD D 4 Bright A_CustomMissile("CacodemonBall_Base", 24, 0, random(-35, 35))
			Goto See
		Pain.Mightyboot:
			TNT1 A 0 A_Facetarget
			TNT1 A 0 ThrustThing(angle*256/360+128, 10, 0, 0)
			Goto Pain
		Pain:
			HEAD E 3
			HEAD E 3 A_Pain
			HEAD F 6
			Goto See
		XDeath: // Signal to ACS code that we can gib this actor.
		Death:
			HEAD G 8
			HEAD H 8 A_Scream
			HEAD IJ 8
			HEAD K 8 A_NoBlocking
			HEAD L -1 A_SetFloorClip
			Stop
		Ice:
			HEAD E 5 A_GenericFreezeDeath
			HEAD E 1 A_FreezeDeathChunks
			wait
		Raise:
			HEAD KJIHG 5
			Goto See
	}
}
