ACTOR SH_Revenant : SH_BaseMonster
{
	Tag "Revenant"
	Species "Revenant"
	Health 300
	Radius 20
	Height 56
	Mass 500
	Speed 10
	PainChance 100
	MeleeThreshold 196
	+NOICEDEATH
	+MISSILEMORE
	+FLOORCLIP
	SeeSound "skeleton/sight"
	PainSound "skeleton/pain"
	DeathSound "skeleton/death"
	ActiveSound "skeleton/active"
	MeleeSound "skeleton/melee"
	HitObituary "$OB_UNDEADHIT" // "%o was punched by a revenant."
	Obituary "$OB_UNDEAD" // "%o couldn't evade a revenant's fireball."
	States
	{
		Spawn:
			REV1 A 0 NoDelay A_SetUserVar("user_hpbar_offset", 24)
			goto SpawnLoop
		SpawnLoop:
			"####" AB 10 A_Look
			Loop
		See:
			"####" AABBCCDDEEFF 2 A_Chase
			Loop
		Melee:
			"####" G 1 A_FaceTarget
			"####" G 6 A_SkelWhoosh
			"####" H 6 A_FaceTarget
			"####" I 6 A_SkelFist
			Goto See
		Missile:
			"####" J 1 Bright A_FaceTarget
			"####" J 9 Bright A_FaceTarget
			"####" K 10 A_CustomMissile ("RevenantTracer", 50.0, -16, 0, CMF_CHECKTARGETDEAD)
			"####" J 4 Bright A_FaceTarget
			"####" K 10 A_CustomMissile ("RevenantTracer", 50.0, 16, 0, CMF_CHECKTARGETDEAD)
			"####" K 10 A_FaceTarget
			Goto See
		Pain:
			"####" L 5
			"####" L 5 A_Pain
			Goto See
		Death:
			"####" LM 7
			"####" N 7 A_Scream
			"####" O 7 A_NoBlocking
			"####" P 7
			"####" Q -1
			Stop
		Ice:
			"####" L 5 A_GenericFreezeDeath
			"####" L 1 A_FreezeDeathChunks
			wait
		Raise:
			"####" Q 5
			"####" PONML 5
			Goto See
	}
}
