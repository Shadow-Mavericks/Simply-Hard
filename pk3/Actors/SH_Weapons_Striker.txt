actor StrikerBaseWeapon : SH_Weapon
{
	Inventory.RestrictedTo "Striker"
	States
	{
		Abilities:
			TNT1 A 1 A_TakeInventory("AbilityActive", 1)
			TNT1 A 0 A_JumpIfInventory("DualFireHax", 1, "UseAbility")
			loop
		UseAbility:
			TNT1 A 0 A_JumpIfInventory("AuraBallAbility", 1, "AuraBallCheck")
			goto Abilities
		AuraBallCheck:
			TNT1 A 0 A_JumpIf(CallACS("STRIKER_USEAURA", 5), "AuraBall")
			goto Abilities
		AuraBall:
			AAB0 A 0 A_GiveInventory("AbilityActive", 1)
			AAB0 A 0 A_PlaySound("AuraFire", CHAN_AUTO)
			AAB0 AB 1
			AAB0 CDEFGH 1 BRIGHT
			AAB0 I 1 BRIGHT A_FireCustomMissile("AuraBall", 0, 0, -6, 0, 0, 0)
			AAB0 JKLM 1 BRIGHT
			AAB0 OPQR 1
			goto Abilities
			
		HealSelf:
			STSH AB 1
			STSH C 0 A_PlaySound("AuraFire", CHAN_AUTO)
			STSH CDEFGHIJKL 1 BRIGHT
			STSH M 0 BRIGHT A_PlaySound("Striker/SelfHeal", CHAN_AUTO)
			STSH M 0 BRIGHT A_SetBlend("64 00 FF", 0.75, 15, "64 00 FF")
			STSH M 1 BRIGHT ACS_NamedExecuteWithResult("STRIKER_SELFHEAL", 1)
			STSH NOPQR 1 BRIGHT
			goto Abilities
			
		NoAbilityUse:
			TNT1 A 1
			loop
	}
}

ACTOR StrikersFists : StrikerBaseWeapon
{
	Tag "Fists"
	Weapon.SelectionOrder 9
	Weapon.SlotNumber 1
	Inventory.RestrictedTo "Striker"
	Weapon.Kickback 100
	Obituary "%o was beaten to the core by %k"
	+WEAPON.WIMPY_WEAPON
	+WEAPON.MELEEWEAPON
	+WEAPON.CHEATNOTWEAPON
	+WEAPON.NOALERT
	States
	{
	Ready:
		TNT1 A 0 A_GunFlash("Abilities", GFF_NOEXTCHANGE)
		TNT1 A 1 A_WeaponReady(WRF_ALLOWZOOM)
		goto Ready+1
	Deselect:
		TNT1 AAAA 0 A_Lower
		TNT1 A 1 A_Lower
		Loop
	Select:
		TNT1 AAAA 0 A_Raise
		TNT1 A 1 A_Raise
		Loop
	
	Fire:
		TNT1 A 0 A_JumpIfInventory("AbilityActive", 1, "CannotPunch")
		TNT1 A 0 A_GunFlash("NoAbilityUse", GFF_NOEXTCHANGE)
		
		TNT1 A 0 A_PlaySound("skeleton/swing", CHAN_AUTO)
		TNT1 A 0 A_JumpIfInventory("FistSelector", 3, "UppercutRight")
		TNT1 A 0 A_JumpIfInventory("FistSelector", 2, "UppercutLeft")
		TNT1 A 0 A_JumpIfInventory("FistSelector", 1, "SwingRight")
		goto SwingLeft
	CannotPunch:
		TNT1 A 1
		goto Ready+1
	SwingLeft:
		SPUN A 0 A_GiveInventory("FistSelector", 1)
		SPUN AB 1
		SPUN CDE 1 BRIGHT
		goto SwingLeftHitNormal
	SwingLeftHitNormal:
		SPUN F 0 BRIGHT A_JumpIfInventory("PowerStrength", 1, "SwingLeftHitBerserk")
		SPUN F 1 BRIGHT A_CustomPunch(50, true, 0, "StrikersVoidDrainPuff", 64.0)
		SPUN GH 1 BRIGHT
		SPUN IJ 1
		TNT1 A 2
		goto Ready+1
	SwingLeftHitBerserk:
		SPUN F 1 A_CustomPunch(150, true, 0, "StrikersVoidDrainPuff", 64.0)
		SPUN GHIJ 1
		TNT1 A 2
		goto Ready+1
		
	SwingRight:
		SPU2 A 0 A_GiveInventory("FistSelector", 1)
		SPU2 AB 1
		SPU2 CDE 1 BRIGHT
		goto SwingHitRightNormal
	SwingHitRightNormal:
		SPU2 F 0 A_JumpIfInventory("PowerStrength", 1, "SwingRightHitBerserk")
		SPU2 F 1 A_CustomPunch(60, true, 0, "StrikersVoidDrainPuff", 64.0)
		SPU2 GH 1 BRIGHT
		SPU2 IJ 1
		TNT1 A 2
		goto Ready+1
	SwingRightHitBerserk:
		SPU2 F 1 BRIGHT A_CustomPunch(160, true, 0, "StrikersVoidDrainPuff", 64.0)
		SPU2 GH 1 BRIGHT
		SPU2 IJ 1
		TNT1 A 2
		goto Ready+1
	
	UppercutLeft:
		SPU3 A 0 A_GiveInventory("FistSelector", 1)
		SPU3 AB 1
		SPU3 CDE 1 BRIGHT
		goto UppercutLeftHitNormal
	UppercutLeftHitNormal:
		SPU3 F 0 BRIGHT A_JumpIfInventory("PowerStrength", 1, "UppercutLeftHitBerserk")
		SPU3 F 1 BRIGHT A_CustomPunch(60, true, 0, "StrikersUppercut", 64.0)
		SPU3 GH 1 BRIGHT
		SPU3 IJ 1
		TNT1 A 2
		goto Ready+1
	UppercutLeftHitBerserk:
		SPU3 F 1 A_CustomPunch(160, true, 0, "StrikersUppercut", 64.0)
		SPU3 GHIJ 1
		TNT1 A 2
		goto Ready+1
		
	UppercutRight:
		SPU4 A 0 A_TakeInventory("FistSelector", 3)
		SPU4 AB 1
		SPU4 CDE 1 BRIGHT
		goto UppercutRightHitNormal
	UppercutRightHitNormal:
		SPU4 F 0 BRIGHT A_JumpIfInventory("PowerStrength", 1, "UppercutRightHitBerserk")
		SPU4 F 1 BRIGHT A_CustomPunch(70, true, 0, "StrikersUppercut", 64.0)
		SPU4 GH 1 BRIGHT
		SPU4 IJ 1
		TNT1 A 2
		goto Ready+1
	UppercutRightHitBerserk:
		SPU4 F 1 A_CustomPunch(170, true, 0, "StrikersUppercut", 64.0)
		SPU4 GHIJ 1
		TNT1 A 2
		goto Ready+1
		
	Zoom:
	HealCheck:
		"####" A 0 A_JumpIf(CallACS("STRIKER_SELFHEAL", 0), "HealTrigger")
		"####" A 1
		goto Ready+1
	HealTrigger:
		"####" A 18 A_GunFlash("HealSelf", GFF_NOEXTCHANGE)
		goto Ready
		
	NoAbilityUse:
		TNT1 A 12
		goto Abilities
	}

		Deselect:

			PPKS A 1 A_Lower
			Goto Deselect
		Select:
			PPKS AAA 0 A_Raise 
			PPKS A 1 A_Raise
			Goto Select
		Fire:
			PPKS A 0 A_JumpIfNoAmmo("ReloadOrDryfire")
			PPKS A 0 A_JumpIfInventory("PistolImbued", 1, "ImbuedFire")
			goto NormalFire
		NormalFire:
			PPKS A 0 A_PlaySound("WaltherPPK", CHAN_WEAPON)

			PPKS B 0 BRIGHT A_FireBullets(2, 2, 1, CallACS("STRIKER_PPKDAMAGE"), "SH_PlayerBulletPuff", FBF_USEAMMO | FBF_NORANDOM)

			PPKS C 0 BRIGHT A_Light1
			PPKS C 1 BRIGHT A_SetPitch(pitch+0.125, 0)//SPF_INTERPOLATE)
			PPKS A 0 BRIGHT A_Light0
			PPKS A 1 A_SetPitch(pitch+0.125, 0)//SPF_INTERPOLATE)
			PPKS A 3
			PPKS A 0 A_JumpIfNoAmmo("ReloadOrDryfire")
			PPKS A 0 A_Refire
			Goto Ready+1
		ImbuedFire:
			PPKS A 0 A_PlaySound("WaltherPPK", CHAN_WEAPON)
			PPKS A 0 A_PlaySound("AuraBlast", CHAN_AUTO)	
			PPKS A 0 A_TakeInventory("PistolImbued", 1)
			PPK2 B 0 BRIGHT A_AlertMonsters
			PPK2 B 0 BRIGHT A_FireBullets(2, 2, 1, CallACS("STRIKER_PPKDAMAGE")*2, "StrikersVoidPuff", FBF_USEAMMO | FBF_NORANDOM)
			PPK2 B 0 BRIGHT A_Light2
			PPK2 B 1 BRIGHT A_SetPitch(pitch-0.5, 0)//SPF_INTERPOLATE)
			PPK2 C 0 BRIGHT A_Light1
			PPK2 C 1 BRIGHT A_SetPitch(pitch+0.25, 0)//SPF_INTERPOLATE)
			PPKS A 0 BRIGHT A_Light0
			PPKS A 1 A_SetPitch(pitch+0.25, 0)//SPF_INTERPOLATE)
			PPKS A 3
			PPKS A 0 A_JumpIfNoAmmo("ReloadOrDryfire")
			PPKS A 0 A_Refire
			Goto Ready+1
		ReloadOrDryfire:
			"####" A 0 A_JumpIfInventory("peastolClip", 1, "CheckReload")
			"####" A 0 A_Playsound("weapons/pistol/dry", CHAN_AUTO)
			"####" A 1 Offset(0, 33)
			"####" A 5 Offset(0, 32)
			"####" A 0 A_Refire
			goto Ready+1
			
		Reload:
			PPKS A 0 A_JumpIfInventory("peastolClip", 1, "CheckReload")
			goto Ready+1
		CheckReload:
			PPKS A 0 A_JumpIfInventory("HedgefoxClip", 0, "Ready") // 0 == At Capacity
			goto ReloadStart
		ReloadStart:
			PPKR A 0 A_Playsound("GPistolMagOut", CHAN_AUTO)
			PPKR ABCDEF 1
			goto ReloadHold

		ReloadHold:
			PPKR F 10 A_Playsound("GPistolMagIn", CHAN_AUTO)
			goto ReloadInsert

		ReloadInsert:
			PPKR G 2
			PPKR HI 2
			PPKR I 2
			goto ReloadLoop

		ReloadLoop:
			"####" A 0 A_TakeInventory("peastolClip", 1)
			"####" A 0 A_GiveInventory("HedgefoxClip", 1)
			"####" A 0 A_JumpIfInventory("HedgefoxClip", 0, "ReloadEnd")
			"####" A 0 A_JumpIfInventory("peastolClip", 1, "ReloadLoop")
			goto ReloadEnd

		ReloadEnd:
			PPKR JKLMN 1
			PPKR A 0 A_Refire
			goto Ready+1
			
		Zoom:
			PPKS A 0 A_JumpIfInventory("UseHax", 1, "HealCheck")
			PPKS A 0 A_JumpIfInventory("PistolImbued", 7, "CannotImbue")
			PPKS A 0 A_JumpIf(CallACS("STRIKER_IMBUE", 0), "ImbueTrigger")
			PPKS A 1 A_WeaponReady
			goto Ready+1
		CannotImbue:
			PPKS A 1 A_WeaponReady
			goto Ready+1
		ImbueTrigger:
			PPKS A 28 A_GunFlash("Imbue", GFF_NOEXTCHANGE)
			goto Ready
		Imbue:
			PPKI ABCD 2
			PPKI E 0 A_PlaySound("AuraFire", CHAN_AUTO)
			PPKI EFGHIJK 2 BRIGHT
			PPKI LMN 2
			goto Abilities
			
		HealCheck:
			"####" A 0 A_JumpIf(CallACS("STRIKER_SELFHEAL", 0), "HealTrigger")
			goto CannotImbue
		HealTrigger:
			"####" A 18 A_GunFlash("HealSelf", GFF_NOEXTCHANGE)
			goto Ready

		Spawn:
			PIST A -1
			Stop
	}
