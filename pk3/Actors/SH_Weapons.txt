actor SH_BulletPuff : BulletPuff replaces BulletPuff
{
	Decal BulletChip
	VSpeed 0
	
	RenderStyle Translucent
	Alpha 0.8
	
	+PUFFGETSOWNER
	+NOTIMEFREEZE
	+NEVERADJUSTSPRITECLIPPING
	States
	{
		XDeath:
			PUFF A 2 Bright Light("BPUFF1")
			PUFF B 2 Bright Light("BPUFF2")
			stop
		Spawn:
			PUFF A 2 Bright Light("BPUFF1")
			PUFF B 2 Bright Light("BPUFF2")
			goto Melee
		Melee:
			PUFF C 0 A_ChangeVelocity(0, frandom(-0.5, 0.5), frandom(1.0, 1.5), CVF_REPLACE)
			PUFF C 0 A_SetTranslucent(0.5, 0)
			PUFF CCCC 1 A_SetScale(scalex+0.0625)
			goto Fade
		Fade:
			PUFF D 1 A_SetScale(scalex+0.03125)
			PUFF D 0 A_FadeOut(0.025)
			loop
	}
}

ACTOR UAC-Chainsaw : SH_Weapon
{
	Tag "Chainsaw"
	Obituary "$OB_MPCHAINSAW"
	Weapon.SlotNumber 1
	Weapon.Kickback 0
	Weapon.SelectionOrder 9
	Weapon.UpSound "weapons/sawup"
	Weapon.ReadySound "weapons/sawidle"
	Inventory.PickupMessage "$GOTCHAINSAW"
	Inventory.RestrictedTo "Fiend"
	Inventory.PickupSound "weapon/chainget"
	+WEAPON.MELEEWEAPON
	States
	{
		Ready:
			SAW2 CD 4 A_WeaponReady
			SAW2 CD 4 A_WeaponReady
			Loop
		Deselect:
			SAW2 C 0 A_Playsound("Chainsaw/shutoff",Chan_weapon)
			SAW2 C 1 A_Lower
			Goto Deselect+1
		Select:
			SAW2 CCCCDDDD 1 A_Raise
			Loop
		Fire:
			SAW2 A 0 A_JumpIfInventory("PowerStrength", 1, "Berserked")
			SAW2 AB 2 A_Saw
			SAW2 B 0 A_ReFire
			Goto Ready
		Berserked:
			SAW2 AB 1 A_Saw
			SAW2 B 0 A_ReFire
			Goto Ready
		Spawn:
			2SAW A -1
			Stop
	}
}

		Spawn:
			TOK8 ABCB 2 BRIGHT Light("RedUpgrade")
			"####" A 0 A_ClearTarget
			"####" A 0 A_LookEx(LOF_NOSOUNDCHECK | LOF_NOSEESOUND, 0.0, 128.0, 128.0, 360.0, "TokenCheck")
			loop
		TokenCheck:
			"####" A 0 A_JumpIfInTargetInventory("Tier8UpgradeToken", 1, "Spawn") // Got the token? Back to spawn, which will clear the target.
			"####" A 0 A_RearrangePointers(AAPTR_TARGET, AAPTR_MASTER, AAPTR_TARGET, 0)
			"####" A 0 A_ChangeFlag("NOGRAVITY", 1)
			"####" A 0 A_ChangeFlag("SEEKERMISSILE", 1)
			"####" A 0 A_ChangeFlag("NOCLIP", 1)
			goto SuccLoop
		SuccLoop:
			"####" A 1 Bright Light("RedUpgrade") A_SeekerMissile(11.25, 22.50, SMF_PRECISE, 0, 0)
			"####" A 0 Bright Light("RedUpgrade") A_Weave(4, 4, frandom(-1.0, 1.0), frandom(-1.0, 1.0))
			"####" B 1 Bright Light("RedUpgrade") A_SeekerMissile(11.25, 22.50, SMF_PRECISE, 0, 0)
			"####" B 0 Bright Light("RedUpgrade") A_Weave(4, 4, frandom(-1.0, 1.0), frandom(-1.0, 1.0))
			"####" C 1 Bright Light("RedUpgrade") A_SeekerMissile(11.25, 22.50, SMF_PRECISE, 0, 0)
			"####" C 0 Bright Light("RedUpgrade") A_Weave(4, 4, frandom(-1.0, 1.0), frandom(-1.0, 1.0))
			"####" B 1 Bright Light("RedUpgrade") A_SeekerMissile(11.25, 22.50, SMF_PRECISE, 0, 0)
			"####" B 0 Bright Light("RedUpgrade") A_Weave(4, 4, frandom(-1.0, 1.0), frandom(-1.0, 1.0))
			"####" B 0 Bright A_JumpIfInTargetInventory("Tier8UpgradeToken", 1, "DropToGround")
			loop
		DropToGround:
			"####" A 0 A_ChangeFlag("NOGRAVITY", 0)
			"####" A 0 A_ChangeFlag("SEEKERMISSILE", 0)
			"####" A 0 A_ChangeFlag("NOCLIP", 0)
			goto Spawn
	}
}
